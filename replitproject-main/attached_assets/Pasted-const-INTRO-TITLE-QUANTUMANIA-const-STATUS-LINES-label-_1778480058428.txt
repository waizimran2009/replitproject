const INTRO_TITLE = "QUANTUMANIA"
const STATUS_LINES = [
  { label: "NEURAL NETWORKS", status: "ONLINE",  color: "rgba(52,211,153,0.9)",  delay: "3.8s" },
  { label: "QUANTUM CORE",    status: "ACTIVE",  color: "rgba(52,211,153,0.9)",  delay: "4.2s" },
  { label: "MEMORY MATRIX",   status: "SYNCED",  color: "rgba(52,211,153,0.9)",  delay: "4.6s" },
  { label: "LANGUAGE MODEL",  status: "LOADED",  color: "rgba(251,191,36,0.9)",  delay: "5.0s" },
]
function playIntroSound() {
  try {
    const AC = window.AudioContext || (window as any).webkitAudioContext
    if (!AC) return
    const ctx = new AC()
    const now = ctx.currentTime
    const master = ctx.createGain()
    master.gain.setValueAtTime(0.7, now)
    master.connect(ctx.destination)
    // 1. Deep power-up bass rumble (0s → 2.5s)
    const bass = ctx.createOscillator()
    const bassGain = ctx.createGain()
    bass.type = "sine"
    bass.frequency.setValueAtTime(30, now)
    bass.frequency.exponentialRampToValueAtTime(110, now + 2)
    bassGain.gain.setValueAtTime(0, now)
    bassGain.gain.linearRampToValueAtTime(0.35, now + 0.4)
    bassGain.gain.exponentialRampToValueAtTime(0.001, now + 2.8)
    bass.connect(bassGain); bassGain.connect(master)
    bass.start(now); bass.stop(now + 2.9)
    // 2. Orb whoosh — noise burst + filter sweep (0.2s)
    const bufSize = ctx.sampleRate * 1.5
    const noiseBuffer = ctx.createBuffer(1, bufSize, ctx.sampleRate)
    const noiseData = noiseBuffer.getChannelData(0)
    for (let i = 0; i < bufSize; i++) noiseData[i] = Math.random() * 2 - 1
    const noise = ctx.createBufferSource()
    noise.buffer = noiseBuffer
    const noiseFilter = ctx.createBiquadFilter()
    noiseFilter.type = "bandpass"
    noiseFilter.frequency.setValueAtTime(200, now + 0.2)
    noiseFilter.frequency.exponentialRampToValueAtTime(4000, now + 1.4)
    noiseFilter.Q.setValueAtTime(1.5, now + 0.2)
    const noiseGain = ctx.createGain()
    noiseGain.gain.setValueAtTime(0, now + 0.2)
    noiseGain.gain.linearRampToValueAtTime(0.12, now + 0.35)
    noiseGain.gain.exponentialRampToValueAtTime(0.001, now + 1.6)
    noise.connect(noiseFilter); noiseFilter.connect(noiseGain); noiseGain.connect(master)
    noise.start(now + 0.2); noise.stop(now + 1.8)
    // 3. Scan ring pings (3 short tones)
    const ringPings = [
      { t: 0.55, freq: 660,  dur: 0.6 },
      { t: 0.95, freq: 880,  dur: 0.6 },
      { t: 1.35, freq: 1100, dur: 0.6 },
    ]
    ringPings.forEach(({ t, freq, dur }) => {
      const osc = ctx.createOscillator()
      const g = ctx.createGain()
      osc.type = "sine"
      osc.frequency.setValueAtTime(freq, now + t)
      osc.frequency.exponentialRampToValueAtTime(freq * 0.7, now + t + dur)
      g.gain.setValueAtTime(0, now + t)
      g.gain.linearRampToValueAtTime(0.08, now + t + 0.03)
      g.gain.exponentialRampToValueAtTime(0.001, now + t + dur)
      osc.connect(g); g.connect(master)
      osc.start(now + t); osc.stop(now + t + dur + 0.1)
    })
    // 4. Title reveal chime — C major chord (2.5s)
    const chordNotes = [
      { freq: 523.25, gain: 0.15 },
      { freq: 659.25, gain: 0.10 },
      { freq: 783.99, gain: 0.08 },
    ]
    chordNotes.forEach(({ freq, gain }) => {
      const osc = ctx.createOscillator()
      const g = ctx.createGain()
      osc.type = "sine"
      osc.frequency.setValueAtTime(freq, now + 2.5)
      g.gain.setValueAtTime(0, now + 2.5)
      g.gain.linearRampToValueAtTime(gain, now + 2.55)
      g.gain.exponentialRampToValueAtTime(0.001, now + 4.0)
      osc.connect(g); g.connect(master)
      osc.start(now + 2.5); osc.stop(now + 4.1)
    })
    // 5. High shimmer sweep after title (2.6s)
    const shimmer = ctx.createOscillator()
    const shimmerGain = ctx.createGain()
    shimmer.type = "sawtooth"
    shimmer.frequency.setValueAtTime(3000, now + 2.6)
    shimmer.frequency.exponentialRampToValueAtTime(800, now + 3.4)
    shimmerGain.gain.setValueAtTime(0, now + 2.6)
    shimmerGain.gain.linearRampToValueAtTime(0.03, now + 2.7)
    shimmerGain.gain.exponentialRampToValueAtTime(0.001, now + 3.4)
    shimmer.connect(shimmerGain); shimmerGain.connect(master)
    shimmer.start(now + 2.6); shimmer.stop(now + 3.5)
    // 6. Status line blips (one per line)
    const blipTimes = [3.8, 4.2, 4.6, 5.0]
    blipTimes.forEach(t => {
      const osc = ctx.createOscillator()
      const g = ctx.createGain()
      osc.type = "square"
      osc.frequency.setValueAtTime(1200, now + t)
      osc.frequency.exponentialRampToValueAtTime(600, now + t + 0.08)
      g.gain.setValueAtTime(0.04, now + t)
      g.gain.exponentialRampToValueAtTime(0.001, now + t + 0.12)
      osc.connect(g); g.connect(master)
      osc.start(now + t); osc.stop(now + t + 0.15)
    })
    // 7. Ambient hum throughout (0s → 7s)
    const hum = ctx.createOscillator()
    const humGain = ctx.createGain()
    hum.type = "sine"
    hum.frequency.setValueAtTime(55, now)
    humGain.gain.setValueAtTime(0, now)
    humGain.gain.linearRampToValueAtTime(0.06, now + 1.5)
    humGain.gain.setValueAtTime(0.06, now + 5.5)
    humGain.gain.linearRampToValueAtTime(0, now + 7.0)
    hum.connect(humGain); humGain.connect(master)
    hum.start(now); hum.stop(now + 7.1)
    // 8. Fade out whoosh (6.0s)
    const outro = ctx.createOscillator()
    const outroGain = ctx.createGain()
    outro.type = "sine"
    outro.frequency.setValueAtTime(440, now + 6.0)
    outro.frequency.exponentialRampToValueAtTime(80, now + 7.0)
    outroGain.gain.setValueAtTime(0, now + 6.0)
    outroGain.gain.linearRampToValueAtTime(0.12, now + 6.1)
    outroGain.gain.exponentialRampToValueAtTime(0.001, now + 7.0)
    outro.connect(outroGain); outroGain.connect(master)
    outro.start(now + 6.0); outro.stop(now + 7.1)
  } catch (_) { /* AudioContext not available */ }
}
